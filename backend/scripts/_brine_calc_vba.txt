Private Sub Befehl115_Click()
    Dim Antwort As Long
    Dim VHCl, cHCl, rhoHCl As Double
    Dim rst1, rst2 As DAO.Recordset
    Set rst1 = CurrentDb.OpenRecordset("Select * From tblGewDaten ORDER BY IDSprache")
    GewSprache = rst1!IDSprache
    rst1.Close

    If (IsNull(Me!Text100) And IsNull(Me!Feld60)) Then
        If GewSprache = 1 Then
            MsgBox ("Zuerst Elektrolyseurbezeichnung und Datum eingeben.")
        Else
            MsgBox ("Electrolyser designation and date have to be entered first.")
        End If
        Exit Sub
    ElseIf IsNull(Me!Text100) Then
        If GewSprache = 1 Then
            MsgBox ("Zuerst Elektrolyseurbezeichnung eingeben.")
        Else
            MsgBox ("Electrolyser designation has to be entered first.")
        End If
        Exit Sub
    ElseIf IsNull(Me!Feld60) Then
        If GewSprache = 1 Then
            MsgBox ("Zuerst Datum eingeben.")
        Else
            MsgBox ("Date has to be entered first.")
        End If
        Exit Sub
    End If
    
    Set rst2 = CurrentDb.OpenRecordset(" SELECT * FROM tblAnalyseHClElektrolyseur WHERE" & _
        "(((tblAnalyseHClElektrolyseur.Elektrolyseur)='" & Me!Text100 & "')" & _
        "AND ((tblAnalyseHClElektrolyseur.Datum)=" & Format$(Me!Feld60, "\#m\/d\/yy\#") & ")) ")
        
    If Not (rst2.BOF And rst2.EOF) Then
        rst2.MoveFirst
        If IsNull(rst2![flow rate]) Then
            If GewSprache = 1 Then
                MsgBox ("Keine Daten für HCl Durchfluss gefunden. HCl Konzentration nicht berechnet.")
            Else
                MsgBox ("No HCl flow rate found. HCl concentration not calculated.")
            End If
            Exit Sub
        Else
            VHCl = rst2![flow rate]
        End If
        If IsNull(rst2!HCl) Then
            If GewSprache = 1 Then
                MsgBox ("Keine Daten für HCl Konzentration gefunden. HCl Konzentration in Reinstsole nicht berechnet.")
            Else
                MsgBox ("No HCl concentration found. HCl concentration in pure brine not calculated.")
            End If
            Exit Sub
        Else
            cHCl = rst2!HCl
        End If
        If IsNull(rst2!Dichte) Then
            If GewSprache = 1 Then
                MsgBox ("Keine Daten für HCl-Dichte gefunden. HCl Konzentration in Reinstsole nicht berechnet.")
            Else
                MsgBox ("No HCl density found. HCl concentration in pure brine not calculated.")
            End If
            Exit Sub
        Else
            rhoHCl = rst2!Dichte
        End If
    Else
        If GewSprache = 1 Then
            MsgBox ("Keine Daten für HCl Durchfluss, Konzentration und Dichte gefunden. HCl Konzentration nicht berechnet.")
        Else
            MsgBox ("No HCl flow rate, concentration and density found. HCl concentration not calculated.")
        End If
        Exit Sub
    End If
    rst2.Close

    If IsNull(Me![flow rate]) = True Then
        If GewSprache = 1 Then
            MsgBox ("Angabe für Reinstsole-Durchfluss fehlt. HCl Konzentration nicht berechnet.")
        Else
            MsgBox ("Pure brine flow rate missing. HCl concentration not calculated.")
        End If
        Exit Sub
    End If
    
    If IsNull(Me![Na2CO3]) = True Then
        If GewSprache = 1 Then
            MsgBox ("Angabe für Na2CO3-Konzentration fehlt. HCl Konzentration nicht berechnet.")
        Else
            MsgBox ("Na2CO3 concentration missing. HCl concentration not calculated.")
        End If
        Exit Sub
    End If
        
    If IsNull(Me![NaOH]) = True Then
        If GewSprache = 1 Then
            MsgBox ("Angabe für NaOH-Konzentration fehlt. HCl Konzentration nicht berechnet.")
            
        Else
            MsgBox ("NaOH concentration missing. HCl concentration not calculated.")
        End If
        Exit Sub
    End If
            
    If IsNull(Me!Text108) = False Then
        If GewSprache = 1 Then
            Antwort = MsgBox("HCl Konzentration bereits eingegeben oder berechnet. Überschreiben?", vbYesNo)
            
        Else
            Antwort = MsgBox("HCl concentration already given or calculated. Overwrite?", vbYesNo)
        End If
        If Antwort = vbNo Then
            Exit Sub
        End If
    End If
    Me!Text108 = (VHCl * rhoHCl * cHCl / 3647 - 1000 * Me![flow rate] * (Me!NaOH / 40 + 2 * Me!Na2CO3 / 105.99)) * 36.47 / (Me![flow rate] * 1000 + VHCl)
    Me!NaOH = 0
    Me!Na2CO3 = 0
    
End Sub

Private Sub Befehl115_Click()
    Dim Antwort As Long
    Dim VHCl, cHCl, rhoHCl As Double
    Dim rst1, rst2 As DAO.Recordset
    Set rst1 = CurrentDb.OpenRecordset("Select * From tblGewDaten ORDER BY IDSprache")
    GewSprache = rst1!IDSprache
    rst1.Close

    If (IsNull(Me!Text100) And IsNull(Me!Feld60)) Then
        If GewSprache = 1 Then
            MsgBox ("Zuerst Elektrolyseurbezeichnung und Datum eingeben.")
        Else
            MsgBox ("Electrolyser designation and date have to be entered first.")
        End If
        Exit Sub
    ElseIf IsNull(Me!Text100) Then
        If GewSprache = 1 Then
            MsgBox ("Zuerst Elektrolyseurbezeichnung eingeben.")
        Else
            MsgBox ("Electrolyser designation has to be entered first.")
        End If
        Exit Sub
    ElseIf IsNull(Me!Feld60) Then
        If GewSprache = 1 Then
            MsgBox ("Zuerst Datum eingeben.")
        Else
            MsgBox ("Date has to be entered first.")
        End If
        Exit Sub
    End If
    
    Set rst2 = CurrentDb.OpenRecordset(" SELECT * FROM tblAnalyseHClElektrolyseur WHERE" & _
        "(((tblAnalyseHClElektrolyseur.Elektrolyseur)='" & Me!Text100 & "')" & _
        "AND ((tblAnalyseHClElektrolyseur.Datum)=" & Format$(Me!Feld60, "\#m\/d\/yy\#") & ")) ")
        
    If Not (rst2.BOF And rst2.EOF) Then
        rst2.MoveFirst
        If IsNull(rst2.[flow rate]) Then
            If GewSprache = 1 Then
                MsgBox ("Keine Daten für HCl Durchfluss gefunden. HCl Konzentration nicht berechnet.")
            Else
                MsgBox ("No HCl flow rate found. HCl concentration not calculated.")
            End If
            Exit Sub
        Else
            VHCl = rst2.[flow rate]
        End If
        If IsNull(rst2.HCl) Then
            If GewSprache = 1 Then
                MsgBox ("Keine Daten für HCl Konzentration gefunden. HCl Konzentration in Reinstsole nicht berechnet.")
            Else
                MsgBox ("No HCl concentration found. HCl concentration in pure brine not calculated.")
            End If
            Exit Sub
        Else
            cHCl = rst2.HCl
        End If
        If IsNull(rst2.Dichte) Then
            If GewSprache = 1 Then
                MsgBox ("Keine Daten für HCl-Dichte gefunden. HCl Konzentration in Reinstsole nicht berechnet.")
            Else
                MsgBox ("No HCl density found. HCl concentration in pure brine not calculated.")
            End If
            Exit Sub
        Else
            rhoHCl = rst2.Dichte
        End If
    Else
        If GewSprache = 1 Then
            MsgBox ("Keine Daten für HCl Durchfluss, Konzentration und Dichte gefunden. HCl Konzentration nicht berechnet.")
        Else
            MsgBox ("No HCl flow rate, concentration and density found. HCl concentration not calculated.")
        End If
        Exit Sub
    End If
    rst2.Close

    If IsNull(Me![flow rate]) = True Then
        If GewSprache = 1 Then
            MsgBox ("Angabe für Reinstsole-Durchfluss fehlt. HCl Konzentration nicht berechnet.")
        Else
            MsgBox ("Pure brine flow rate missing. HCl concentration not calculated.")
        End If
        Exit Sub
    End If
    
    If IsNull(Me![Na2CO3]) = True Then
        If GewSprache = 1 Then
            MsgBox ("Angabe für Na2CO3-Konzentration fehlt. HCl Konzentration nicht berechnet.")
        Else
            MsgBox ("Na2CO3 concentration missing. HCl concentration not calculated.")
        End If
        Exit Sub
    End If
        
    If IsNull(Me![NaOH]) = True Then
        If GewSprache = 1 Then
            MsgBox ("Angabe für NaOH-Konzentration fehlt. HCl Konzentration nicht berechnet.")
            
        Else
            MsgBox ("NaOH concentration missing. HCl concentration not calculated.")
        End If
        Exit Sub
    End If
            
    If IsNull(Me!Text108) = False Then
        If GewSprache = 1 Then
            Antwort = MsgBox("HCl Konzentration bereits eingegeben oder berechnet. Überschreiben?", vbYesNo)
            
        Else
            Antwort = MsgBox("HCl concentration already given or calculated. Overwrite?", vbYesNo)
        End If
        If Antwort = vbNo Then
            Exit Sub
        End If
    End If
    Me!Text108 = (VHCl * rhoHCl * cHCl / 3647 - 1000 * Me![flow rate] * (Me!NaOH / 40 + 2 * Me!Na2CO3 / 105.99)) * 36.47 / (Me![flow rate] * 1000 + VHCl)
    Me!NaOH = 0
    Me!Na2CO3 = 0
    
End Sub

Public Function VAnolytBer(VPureBrine, CNa2SO4PureBrine, CNa2SO4Anolyt) As Double
 If (IsNull(VPureBrine) = True) Or (IsNull(CNa2SO4PureBrine) = True) Or (IsNull(CNa2SO4Anolyt) = True) Then
     VAnolytBer = -1
 Else
    VAnolytBer = VPureBrine * (CNa2SO4PureBrine / CNa2SO4Anolyt)
 End If
End Function
Public Function EtaNaClO3Ber(CNaClO3Anolyt, VAnolyt, CNaClO3PureBrine, VPureBrine, n, i) As Double
If (IsNull(CNaClO3Anolyt) = True) Or (IsNull(VAnolyt) = True) Or (IsNull(CNaClO3PureBrine) = True) Or (IsNull(VPureBrine) = True) Or (IsNull(n) = True) Or (IsNull(i) = True) Then
    EtaNaClO3Ber = -1
Else
    EtaNaClO3Ber = ((CNaClO3Anolyt * VAnolyt) - (CNaClO3PureBrine * VPureBrine)) / (n * i * 0.662) * 100
End If
End Function
Public Function EtaNaOClBer(CNaClOAnolyt, VAnolyt, n, i) As Double
If (IsNull(CNaClOAnolyt) = True) Or (IsNull(VAnolyt) = True) Or (IsNull(n) = True) Or (IsNull(i) = True) Then
    EtaNaOClBer = -1
Else
    If GewClOAnalyse = 1 Then
        EtaNaOClBer = (CNaClOAnolyt * VAnolyt) / (n * i * 1.38869) * 100
    Else
        EtaNaOClBer = (CNaClOAnolyt * VAnolyt) / (n * i * 0.98) * 100
    End If
End If
End Function
Public Function EtaHClBer(CHClPureBrine, VPureBrine, CHClAnolyt, VAnolyt, n, i) As Double
If GewAnsaeurung <> 1 Then
    EtaHClBer = 0
Else
    If (IsNull(CHClPureBrine) = True) Or (IsNull(VPureBrine) = True) Or (IsNull(CHClAnolyt) = True) Or (IsNull(VAnolyt) = True) Or (IsNull(n) = True) Or (IsNull(i) = True) Then
        EtaHClBer = -1
    Else
        EtaHClBer = ((CHClPureBrine * VPureBrine) - (CHClAnolyt * VAnolyt)) / (n * i * 1.36) * 100
    End If
End If
End Function

Public Function EtaAlkaliBer(CNaOHPureBrine, VPureBrine, n, i) As Double
If (IsNull(CNaOHPureBrine) = True) Or (IsNull(VPureBrine) = True) Or (IsNull(n) = True) Or (IsNull(i) = True) Then
    EtaAlkaliBer = -1
Else
    EtaAlkaliBer = (CNaOHPureBrine * VPureBrine) / (n * i * 1.4923) * 100
End If
End Function

Public Function EtaNa2CO3Ber(CNa2CO3PureBrine, VPureBrine, n, i) As Double
If (IsNull(CNa2CO3PureBrine) = True) Or (IsNull(VPureBrine) = True) Or (IsNull(n) = True) Or (IsNull(i)) Then
    EtaNa2CO3Ber = -1
Else
    EtaNa2CO3Ber = (CNa2CO3PureBrine * VPureBrine) / (n * i * 1.978) * 100
End If
End Function

Public Function EtaO2Ber(YO2, YCl2, EtaNaClO3, EtaNaClO) As Double
If (IsNull(YO2) = True) Or (IsNull(YCl2) = True) Or (IsNull(EtaNaClO3) = True) Or (IsNull(EtaNaClO) = True) Then
    EtaO2Ber = -1
Else
    EtaO2Ber = 2 * (YO2 - 0.18 + GewAKF) / (YCl2 + 2 * YO2) * (100 - EtaNaClO3 - EtaNaClO)
End If
End Function

Public Function EtaMembraneBer(EtaO2, EtaNaClO3, EtaNaClO, EtaHCl, EtaNa2CO3, EtaAlkali) As Double
If (IsNull(EtaO2) = True) Or (IsNull(EtaNaClO3) = True) Or (IsNull(EtaNaClO) = True) Or (IsNull(EtaHCl) = True) Or (IsNull(EtaNa2CO3) = True) Or (IsNull(EtaAlkali) = True) Then
    EtaMembraneBer = -1
Else
    EtaMembraneBer = 100 - EtaO2 - EtaNaClO3 - EtaNaClO - EtaHCl + EtaNa2CO3 + EtaAlkali
End If
End Function

Public Function SPCBer(Un, CE) As Double
If (IsNull(Un) = True) Or (IsNull(CE) = True) Then
    SPCBer = -1
Else
    SPCBer = Un * 100000 / (1.4923 * CE)
End If
End Function

Public Function UnBer(U, i, t, c) As Double
If (IsNull(U) = True) Or (IsNull(i) = True) Or (IsNull(t) = True) Or (IsNull(c) = True) Then
     UnBer = -1
Else
    UnBer = 2.42 + (GewiRef / i) * (U - 2.42) - (90 - t) * GewtKorrektur + (32 - c) * GewcKorrektur
End If
End Function

Public Function UnBerVgl(U, i, t, c) As Single
If (IsNull(U) = True) Or (IsNull(i) = True) Or (IsNull(t) = True) Or (IsNull(c) = True) Then
     UnBerVgl = -1
Else
    UnBerVgl = 2.42 + (3 / i) * (U - 2.42) - (90 - t) * 0.01 + (32 - c) * 0.02
End If
End Function
Public Function AktiRef()
    Dim rst1 As DAO.Recordset
    Dim rst2 As DAO.Recordset
    Set rst1 = CurrentDb.OpenRecordset("Select * From tblUnKorrekturfaktoren ORDER BY iRef")
    Set rst2 = CurrentDb.OpenRecordset("Select * From tblGewDaten ORDER BY IDiRef")
    Do Until rst1.EOF
        If rst1!ID = rst2!IDiRef Then
            AktiRef = rst1!iRef
        End If
        rst1.MoveNext
    Loop
    rst1.Close
    rst2.Close
End Function
