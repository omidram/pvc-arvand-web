=== qryVAnolytGruppe
SELECT tblEingabeUnGruppe.Gruppe, tblEingabeUnGruppe.Datum, tblEingabeUnGruppe.Zeit, VAnolytBer([Analyse Pure Brine Gruppe]![flow rate],[Analyse Pure Brine Gruppe]![Na2SO4 Pb],[Analyse Anolyt Gruppe]![Na2SO4 An]) AS VAnolyt

FROM tblGewDaten, tblEingabeUnGruppe INNER JOIN ([Analyse Pure Brine Gruppe] INNER JOIN [Analyse Anolyt Gruppe] ON ([Analyse Pure Brine Gruppe].Datum = [Analyse Anolyt Gruppe].Datum) AND ([Analyse Pure Brine Gruppe].Gruppe = [Analyse Anolyt Gruppe].Gruppe)) ON (tblEingabeUnGruppe.Datum = [Analyse Pure Brine Gruppe].Datum) AND (tblEingabeUnGruppe.Gruppe = [Analyse Pure Brine Gruppe].Gruppe)

WHERE (((tblEingabeUnGruppe.Datum)>=[tblGewDaten].[Anfangsdatum] And (tblEingabeUnGruppe.Datum)<=[tblGewDaten].[Enddatum]) AND ((VAnolytBer([Analyse Pure Brine Gruppe]![flow rate],[Analyse Pure Brine Gruppe]![Na2SO4 Pb],[Analyse Anolyt Gruppe]![Na2SO4 An]))<>(-1))) OR (((tblEingabeUnGruppe.Gruppe) Like (Select IDGruppe FROM tblGewDaten))) OR (((tblEingabeUnGruppe.Gruppe) Like (Select IDGruppe from tblGewDaten)))

ORDER BY tblEingabeUnGruppe.Datum;
