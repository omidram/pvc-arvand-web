Private Sub Befehl46_Click()
On Error GoTo Err_Befehl46_Click

    Dim stDocName As String
    Dim stLinkCriteria As String

    stDocName = "Betriebsdaten"
    DoCmd.OpenForm stDocName, , , stLinkCriteria
    DoCmd.Close acForm, "Datenerfassung Elektrolyseure"

Exit_Befehl46_Click:
    Exit Sub

Err_Befehl46_Click:
    MsgBox Err.Description
    Resume Exit_Befehl46_Click
    
End Sub

Private Sub Form_Open(Cancel As Integer)
    Dim rst1 As DAO.Recordset
    Set rst1 = CurrentDb.OpenRecordset("Select * From tblGewDaten ORDER BY IDSprache")
    GewSprache = rst1!IDSprache
    rst1.Close
    If GewSprache = 1 Then
        Text13.Caption = "Dateneingabe Normierte Spannung Elektrolyseur"
        Text15.Caption = "Elektrolyseur:"
        Text17.Caption = "Datum:"
        Text19.Caption = "Zeit:"
        Text21.Caption = "I gesamt:"
        Text23.Caption = "Cc:"
        Text25.Caption = "t An:"
        Text27.Caption = "t Ka:"
        Text40.Caption = "U gesamt:"
        Text44.Caption = "Anzahl Elemente:"
        Befehl46.Caption = "Normierte Spannung"
        Bezeichnungsfeld55.Caption = "(aus Katholyttemperatur berechnet)"
        
    Else
        Text13.Caption = "Input Standardized Voltage Electrolyzer"
        Text15.Caption = "Electrolyzer:"
        Text17.Caption = "Date:"
        Text19.Caption = "Time:"
        Text21.Caption = "I total:"
        Text23.Caption = "Cc:"
        Text25.Caption = "t An:"
        Text27.Caption = "t Ka:"
        Text40.Caption = "U total:"
        Text44.Caption = "No. of Elements:"
        Befehl46.Caption = "Standardized Voltage"
        Bezeichnungsfeld55.Caption = "(calculated from catholyte temperature)"
    
    End If
        
    If GetIDAnolyttemperaturMessen() = 1 Then
        Bezeichnungsfeld55.Visible = False
    End If
    If GetIDAnolyttemperaturMessen() <> 1 Then
        Me![t An].Enabled = False
    End If

End Sub

Private Sub t_Ka_AfterUpdate()
    If GetIDAnolyttemperaturMessen() <> 1 Then
        Me![t An] = Me![t Ka] + GetTemperaturkorrekturAnolyt
    End If
End Sub
