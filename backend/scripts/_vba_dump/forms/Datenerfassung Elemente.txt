Private Sub Form_Open(Cancel As Integer)
    Dim rst1 As DAO.Recordset
    Set rst1 = CurrentDb.OpenRecordset("Select * From tblGewDaten ORDER BY IDSprache")
    GewSprache = rst1!IDSprache
    rst1.Close
    If GewSprache = 1 Then
        Text13.Caption = "Dateneingabe Normierte Spannung Elemente"
        Text15.Caption = "Elektrolyseur:"
        Text17.Caption = "Datum:"
        Text19.Caption = "Zeit:"
        Text21.Caption = "I gesamt:"
        Text23.Caption = "Cc:"
        Text25.Caption = "t An:"
        Text27.Caption = "t Ka:"
        Bezeichnungsfeld55.Caption = "(Aus Katholyttemperatur berechnet)"
        Bezeichnungsfeld41.Caption = "Ui: Elementspannung incl. Rücken zu Rücken Spannung"
        Bezeichnungsfeld43.Caption = "Um: Berechnete mittlere Elementspannung"
        Schaltfläche39.Caption = "Normierte Spannung"
        Befehl42.Caption = "Aus Datei einlesen"
        
    Else
        Text13.Caption = "Input Standardized Voltage Elements"
        Text15.Caption = "Electrolyzer:"
        Text17.Caption = "Date:"
        Text19.Caption = "Time:"
        Text21.Caption = "I total:"
        Text23.Caption = "Cc:"
        Text25.Caption = "t An:"
        Text27.Caption = "t Ka:"
        Bezeichnungsfeld55.Caption = "(calculated from catholyte temperature)"
        Bezeichnungsfeld41.Caption = "Ui: Element voltage including back to back voltage"
        Bezeichnungsfeld43.Caption = "Um: Calculated average element voltage"
        Schaltfläche39.Caption = "Standardized Voltage"
        Befehl42.Caption = "Read Data from File"
    
    End If

    If GetIDAnolyttemperaturMessen() = 1 Then
        Bezeichnungsfeld55.Visible = False
    End If
    If GetIDAnolyttemperaturMessen() <> 1 Then
        Me![t An].Enabled = False
    End If

End Sub

Private Sub Schaltfläche39_Click()
On Error GoTo Err_Schaltfläche39_Click

    Dim DocName As String
    Dim LinkCriteria As String

    DocName = "Betriebsdaten"
    DoCmd.OpenForm DocName, , , LinkCriteria
    DoCmd.Close acForm, "Datenerfassung Elemente"

Exit_Schaltfläche39_Click:
    Exit Sub

Err_Schaltfläche39_Click:
    MsgBox Error$
    Resume Exit_Schaltfläche39_Click
    
End Sub

        Me![t An] = Me![t Ka] + GetTemperaturkorrekturAnolyt
    End If
End Sub
