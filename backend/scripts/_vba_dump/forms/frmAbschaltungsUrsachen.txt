Private Sub Form_Open(Cancel As Integer)

    Dim rst1 As DAO.Recordset
    Set rst1 = CurrentDb.OpenRecordset("Select * From tblGewDaten ORDER BY IDSprache")
    GewSprache = rst1!IDSprache
    rst1.Close
    
    If GewSprache = 1 Then
        Text8.Caption = "Abschaltungsursachen"
        Bezeichnungsfeld110.Caption = "Code"
        Bezeichnungsfeld93.Caption = "Code"

        Bezeichnungsfeld117.Caption = "Kategorie"
        
        Befehl115.Caption = "Abschaltungen"

        Text8.Caption = "Shut Down Reasons"
        Bezeichnungsfeld110.Caption = "Code"
        Bezeichnungsfeld93.Caption = "Code"
        Bezeichnungsfeld109.Caption = "Reason"
        Bezeichnungsfeld68.Caption = "Reason"
        Bezeichnungsfeld116.Caption = "Category"
        Bezeichnungsfeld117.Caption = "Category"
        
        Befehl115.Caption = "Shut Down"
    End If


End Sub
