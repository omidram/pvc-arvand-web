'Unterprogramm zum Übertragen von Daten aus der Tabelle NormElek in die Tabelle Normierung
Private Sub Befehl191_Click()
Dim rst1, rst2, rst3, rst12 As DAO.Recordset
Dim IDDateiformat As Long
Dim dbs As Database

Set dbs = CurrentDb
dbs.Execute (" Delete * From tblInfoAnalysenUElement ")
Set rst1 = CurrentDb.OpenRecordset(" Select * from Normierung Where " & _
            " ( (IsNull(Elektrolyseur) = False) And (IsNull(Datum) = False) And ([I Gesamt] = Null) ) ")
Set rst3 = CurrentDb.OpenRecordset("tblInfoAnalysenUElement")

'Dateiformat ermitteln
'Bei altem Dateiformat entsprechendes Unterprogramm aufrufen
Set rst12 = CurrentDb.OpenRecordset("Select * From tblGewDaten ORDER BY IDDateiformatUElemente")
IDDateiformat = rst12!IDDateiformatUElemente
rst12.Close
    If IDDateiformat = 1 Then
        AltesDateiformatDatenUebertragen
    Exit Sub
End If



'Sprache bestimmen
    Set rst12 = CurrentDb.OpenRecordset("Select * From tblGewDaten ORDER BY IDSprache")
    GewSprache = rst12!IDSprache
    rst12.Close
    
Do Until rst1.EOF
    Set rst2 = CurrentDb.OpenRecordset(" SELECT * FROM NormElek WHERE " & _
        "(((NormElek.Elektrolyseur)='" & rst1!Elektrolyseur & "')" & _
        "AND ((NormElek.Datum)=" & Format$(rst1!Datum, "\#m\/d\/yy\#") & ")) ")
        
   '     Debug.Print rst2.Elektrolyseur, rst2.Datum
   '     "( IsNull(NormElek.Elektrolyseur)=False ) And ( IsNull(NormElek.Datum) = False) And " & _

   ' Set rst2 = CurrentDb.OpenRecordset(" SELECT * FROM NormElek WHERE" & _
   '     "(((NormElek.Elektrolyseur)='" & rst1!Elektrolyseur & "')" & _
   '     "AND ((NormElek.Datum)=" & Format$(rst1!Datum, "\#m\/d\/yy\#") & ")) ")
       
        
        
    If Not (rst2.BOF And rst2.EOF) Then
        rst2.MoveFirst
        rst1.Edit
        rst1![I Gesamt] = rst2![I Gesamt]
        rst1!Cc = rst2!Cc
        rst1![t An] = rst2![t An]
        rst1![t Ka] = rst2![t Ka]
        rst1!Uo = rst2!Uo
        rst1![U Gesamt] = rst2![U Gesamt]
        rst1.Update
        rst3.AddNew
        rst3!Elektrolyseur = rst1!Elektrolyseur
        rst3!Datum = rst1!Datum
        If GewSprache = 1 Then
            rst3!DatenUebertragen = "Analysen übertragen"
        Else
            rst3!DatenUebertragen = "Analytical data transferred"
        End If
        rst3.Update
    Else
        rst3.AddNew
        rst3!Elektrolyseur = rst1!Elektrolyseur
        rst3!Datum = rst1!Datum
        If GewSprache = 1 Then
            rst3!DatenUebertragen = "Keine Analysen gefunden"
        Else
            rst3!DatenUebertragen = "No analytical Data found"
        End If
        rst3.Update
    End If
    subfrmInfoUebertrageneAnalysen.Requery
    rst2.Close
    rst1.MoveNext
Loop
rst1.Close
rst3.Close
If GewSprache = 1 Then
    MsgBox "Analysedaten für Einzelelement-Normierung aktualisiert"
Else
    MsgBox "Analytical data for standardizsation of single element voltages updated"
End If
End Sub

    Set rst3 = CurrentDb.OpenRecordset("Select * From Spannungen", dbOpenDynaset)
    Set rst4 = CurrentDb.OpenRecordset("Select * From Normierung", dbOpenDynaset)

    GewSprache = rst12!IDSprache
    rst12.Close
        
    dbs.Execute (" Delete * From tblInfoEingeleseneSpannungen ")
    dbs.Execute (" Delete * From tblInfoDateienUElement ")
    subfrmInfoDateienUElemente.Requery
    subfrmInfoEingeleseneUElemente.Requery
    Set rst5 = CurrentDb.OpenRecordset("Select * From tblInfoDateienUElement")
    Set rst6 = CurrentDb.OpenRecordset("Select * From tblInfoEingeleseneSpannungen")

    Set rst7 = CurrentDb.OpenRecordset("Select * From tblGewDaten ORDER BY VerzeichnisUElemente")
    GewVerzeichnisUElemente = rst7!VerzeichnisUElemente
    rst7.Close
    
    'Prüfen, welche Dateien vorhanden sind
    Do Until rst1.EOF
        strDateiname = GewVerzeichnisUElemente & "\" & rst1!Dateiname & "." & rst1!Erweiterung
        If FileExist(strDateiname) Then
            rst5.AddNew
            With rst5
                !Elektrolyseur = rst1!Elektrolyseur
                !Datei = strDateiname
                !Dateiname = rst1!Dateiname
            End With
            rst5.Update

        End If
        rst1.MoveNext

    If (rst5.EOF And rst5.BOF) Then
        If GewSprache = 1 Then
            MsgBox "Keine Dateien gefunden", vbInformation, "Hinweis"
        Else
            MsgBox "No files found", vbInformation, "Remark"
        End If
        rst5.Close
        rst1.Close
        Exit Sub
    Else
        rst5.MoveFirst
        rst1.Close
    End If
    'Einlesen der Einzelelementspannungen aus den gefundenen Dateien
    Do Until rst5.EOF
        Einlesedatei = rst5!Datei
        Debug.Print rst5!Datei, Einlesedatei
        dbs.Execute (" Delete * From tblEingeleseneUElementeAlt ")
        DoCmd.TransferSpreadsheet acImport, acSpreadsheetTypeExcel97, "tblEingeleseneUElementeAlt", Einlesedatei, True
        Set rst2 = CurrentDb.OpenRecordset("Select * From " & "tblEingeleseneUElementeAlt", dbOpenDynaset)
        
        'Vor Einlesen prüfen ob Daten bereits vorhanden sind
        Set rst8 = CurrentDb.OpenRecordset(" SELECT * FROM Spannungen WHERE" & _
        "(((Spannungen.Elektrolyseur)='" & rst5!Elektrolyseur & "')" & _
        "AND ((Spannungen.Datum)=" & Format$(CDate(rst2!Datum), "\#m\/d\/yy\#") & ")" & _
        "AND ((Spannungen.Zeit)=#12/30/1899 " & rst2!Zeit & "#)) ")

        If Not (rst8.BOF And rst8.EOF) Then
            If GewSprache = 1 Then
                MsgBox "Die Daten für Elektrolyseur " & rst5!Elektrolyseur & " vom " & rst2!Datum & " um " & _
                rst2!Zeit & " sind bereits vorhanden." & vbCrLf & vbCrLf & _
                "Diese Daten werden nicht eingelesen.", vbInformation, "Hinweis"
            Else
                MsgBox "Data for electrolyser " & rst5!Elektrolyseur & " from " & rst2!Datum & " at " & _
                rst2!Zeit & " already existing." & vbCrLf & vbCrLf & _
                "These data will not be transferred.", vbInformation, "Remark"
            End If
        Else
            'Datensatz in Tabelle Normierung anlegen
            With rst4
                .AddNew
                !Elektrolyseur = rst5!Elektrolyseur
                !Datum = rst2!Datum
                !Zeit = rst2!Zeit
                ![I Gesamt] = rst2![I Ges]
                ![t Ka] = rst2![tKa]
                If GetIDAnolyttemperaturMessen() <> 1 Then
                    ![t An] = rst2![tKa] + GetTemperaturkorrekturAnolyt
                End If
                .Update
            End With
            
            'Datensatz mit Identifikationsnummer für Einzelelemente finden
            Set rst9 = CurrentDb.OpenRecordset(" SELECT * FROM Normierung WHERE" & _
                        "(((Normierung.Elektrolyseur)='" & rst5!Elektrolyseur & "')" & _
                        "AND ((Normierung.Datum)=" & Format$(CDate(rst2!Datum), "\#m\/d\/yy\#") & ")" & _
                        "AND ((Normierung.Zeit)=#12/30/1899 " & rst2!Zeit & "#)) ")

            'Einzelelementspannungen aus Einlesedatei in Tabelle Spannungen übertragen
            'Spaltennummerierung in rst2 beginnt bei 0
            For i = 1 To AnzahlElemente
                Set rst12 = CurrentDb.OpenRecordset(" SELECT * FROM tblReserveplaetze WHERE" & _
                        "((tblReserveplaetze.Position)=" & i & ")  ")
                'Daten werden nur dann übertragen, wenn kein Reserveplatz vorliegt oder Reserveplätze
                'eingelesen werden
                If (rst12.EOF And rst12.BOF) Or (GetIDReserveplaetzeEinlesen() = 1) Then
                    With rst3
                        .AddNew
                        !Normierung = rst9.Normierung
                        !Elektrolyseur = rst5!Elektrolyseur
                        !Datum = rst2!Datum
                        !Zeit = rst2!Zeit
                        !Position = i
                        '!Ui = Einzelspannung(i)
                        !Ui = rst2(i + 4)
                        .Update
                    End With
                End If
                rst12.Close
            Next
            
            'Datensatz in Infotabelle für eingelesene Spannungen anlegen
            With rst6
                .AddNew
                !Elektrolyseur = rst5!Elektrolyseur
                !Datum = rst2!Datum
                !Zeit = rst2!Zeit
                .Update
            End With
            rst9.Close
        End If
        subfrmInfoEingeleseneUElemente.Requery
        rst2.Close
        rst8.Close
        rst5.MoveNext
    Loop
    rst3.Close
    rst4.Close
    rst5.Close
    rst6.Close
    If GewSprache = 1 Then
        MsgBox "Einlesen der Einzelelementspannungen abgeschlossen."
    Else
        MsgBox "Reading of single element voltages from file completed."
    End If
End Sub
