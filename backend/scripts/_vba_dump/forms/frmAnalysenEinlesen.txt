Private Sub AnalysenReinstsoleEinlesen()
    Dim db, dbs As Database
    Dim td As TableDef
    Dim rst1, rst2, rst3, rst4, rst5, rst6, rst7, rst8, rst9, rst10, rst11, rst12 As DAO.Recordset
    Dim Einlesedatei As Variant
    Dim i, neueID, AnzahlElemente, IDDateiformat As Long
    Dim fld As DAO.Field
    Dim GewVerzeichnisAnalysen, GewDateinameAnalysen, strDateiname As String
    
    
    Set dbs = CurrentDb
    
    'Set rst1 = CurrentDb.OpenRecordset("Select * From tblDateinamenUElement", dbOpenDynaset)
    'Set rst3 = CurrentDb.OpenRecordset("Select * From [Analyse Anolyt]", dbOpenDynaset)
    'Set rst4 = CurrentDb.OpenRecordset("Select * From Normierung", dbOpenDynaset)
    
    'Anzahl der einzulesenden Einzelelementspannungen aus Tabelle tblGewDaten ermitteln
    'Set rst11 = CurrentDb.OpenRecordset("Select * From tblGewDaten ORDER BY AnzahlUElemente")
    'AnzahlElemente = rst11!AnzahlUElemente
    'rst11.Close
    
    'Sprache ermitteln
    Set rst12 = CurrentDb.OpenRecordset("Select * From tblGewDaten ORDER BY IDSprache")
    GewSprache = rst12!IDSprache
    rst12.Close
        
    dbs.Execute (" Delete * From tblInfoEingeleseneAnalysen ")
    dbs.Execute (" Delete * From tblInfoDateienAnalysen ")
    subfrmInfoDateienAnalysen.Requery
    subfrmInfoEingeleseneAnalysen.Requery
    Set rst5 = CurrentDb.OpenRecordset("Select * From tblInfoDateienAnalysen")
    Set rst6 = CurrentDb.OpenRecordset("Select * From tblInfoEingeleseneAnalysen")
    
    ' Verzeichnis für die Datei mit Analysen einlesen
    Set rst7 = CurrentDb.OpenRecordset("Select * From tblGewDaten ORDER BY VerzeichnisAnalysen")
    GewVerzeichnisAnalysen = rst7!VerzeichnisAnalysen
    GewDateinameAnalysen = rst7!DateinameAnalysenReinstsole
    rst7.Close
    
    'Prüfen, ob Datei vorhanden ist
    strDateiname = GewVerzeichnisAnalysen & "\" & GewDateinameAnalysen
    If FileExist(strDateiname) Then
        rst5.AddNew
        With rst5
            '!Elektrolyseur = rst1!Elektrolyseur
            !Datei = strDateiname
            '!Dateiname = rst1!Dateiname
            '!Dateityp = rst1!Dateityp
        End With
        rst5.Update
        subfrmInfoDateienAnalysen.Requery
    End If
        
    If (rst5.EOF And rst5.BOF) Then
        If GewSprache = 1 Then
            MsgBox "Keine Dateien gefunden", vbInformation, "Hinweis"
        Else
            MsgBox "No files found", vbInformation, "Remark"
        End If
        rst5.Close
        'rst1.Close
        Exit Sub
    Else
        rst5.MoveFirst
        'rst1.Close
    End If
    
    'Einlesen der Reinstsoleanalysen aus der gefundenen Datei
    Einlesedatei = strDateiname
    'Debug.Print Einlesedatei
    dbs.Execute (" Delete * From tblEingeleseneAnalysenReinstsole ")
    DoCmd.TransferSpreadsheet acImport, acSpreadsheetTypeExcel97, "tblEingeleseneAnalysenReinstsole", Einlesedatei, True
    Set rst2 = CurrentDb.OpenRecordset("Select * From tblEingeleseneAnalysenReinstsole Where Description = 'Puurpekel stroomefficiëntie' ", dbOpenDynaset)
        
    If rst2.BOF And rst2.EOF Then
        Exit Sub
    End If
        
    rst2.MoveFirst
        
    Do Until rst2.EOF
        'Vor Einlesen prüfen ob Daten bereits vorhanden sind
        Set rst8 = CurrentDb.OpenRecordset(" SELECT * FROM [Analyse pure brine] WHERE" & _
        "((([Analyse pure brine].Elektrolyseur)='" & rst2!Klantref & "')" & _
        "AND (([Analyse pure brine].Datum)=" & Format$(CDate(rst2!Datum), "\#m\/d\/yy\#") & ")" & _
        ") ")

        '  Set rst3 = CurrentDb.OpenRecordset(" SELECT * FROM tblAnodeninstandhaltung WHERE" & _
        '  "((([tblAnodeninstandhaltung!Anoden Nr])='" & rst2![Anoden Nr] & "')" & _
        '  "AND ((tblAnodeninstandhaltung!Versanddatum)=" & Format$(CDate(rst2!Versanddatum), "\#m\/d\/yy\#") & ")" & _
        '   ") ")


        If Not (rst8.BOF And rst8.EOF) Then
        '    If GewSprache = 1 Then
        '        MsgBox "Die Daten für Elektrolyseur " & rst5!Elektrolyseur & " vom " & rst2!Datum & " um " & _
        '        rst2!Zeit & " sind bereits vorhanden." & vbCrLf & vbCrLf & _
        '        "Diese Daten werden nicht eingelesen.", vbInformation, "Hinweis"
        '    Else
        '        MsgBox "Data for electrolyser " & rst5!Elektrolyseur & " from " & rst2!Datum & " at " & _
        '        rst2!Zeit & " already existing." & vbCrLf & vbCrLf & _
        '        "These data will not be transferred.", vbInformation, "Remark"
        '    End If
        Else
            'Datensatz in Tabelle Analyse Anolyt anlegen
            With rst8
                .AddNew
                !Elektrolyseur = rst2!Klantref
                !Datum = rst2!Datum
                '!Zeit = rst2!Zeit
                ![NaClO3 Pb] = rst2![NaClO3(g/l)]
                ![Na2SO4 Pb] = rst2![Na2SO4(g/l)]
                ![NaOCl Pb] = rst2![HOCL(mg/l)] / 1000
                ![HCl Pb] = rst2![HCl(mg/l)] / 1000
                .Update
            End With
            
            
            'Datensatz in Infotabelle für eingelesene Analysen anlegen
            With rst6
                .AddNew
                !Elektrolyseur = rst2!Klantref
                !Datum = rst2!Datum
                '!Zeit = rst2!Zeit
                .Update
            End With
            
        End If
        subfrmInfoEingeleseneAnalysen.Requery
        rst8.Close
        rst2.MoveNext
    Loop
    rst2.Close
    
    If GewSprache = 1 Then
        MsgBox "Einlesen der Reinstsoleanalysen abgeschlossen."
    Else
        MsgBox "Reading of pure brine analyses from file completed."
    End If
End Sub
