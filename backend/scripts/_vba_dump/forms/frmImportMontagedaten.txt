    Set rst2 = CurrentDb.OpenRecordset("Select * From tblImportMontage WHERE [Element Nr] is not Null ORDER BY [Element Nr]")
    
    'Einlesen der Anodeninstandhaltung-Daten aus der gefundenen Datei

            If (rst3.BOF And rst3.EOF) Then
                 With rst3

                    ![Element Nr] = rst2![Element Nr]
                    ![Anoden Nr] = rst2![Anoden Nr]
                    ![Kathoden Nr] = rst2![Kathoden Nr]

                    ![Abschaltgrund] = rst2![Abschaltgrund]
                    ![Anode coating] = rst2![Anode coating]
                    ![Anode electrode] = rst2![Anode electrode]
                    ![Anode shell] = rst2![Anode shell]
                    ![Cathode coating] = rst2![Cathode coating]
                    ![Cathode electrode] = rst2![Cathode electrode]
                    ![Cathode shell] = rst2![Cathode shell]
                    ![Info membrane] = rst2![Info membrane]
                    ![Generation] = rst2![Generation]
                    .Update
                End With
            Else
                With rst3
                    .Edit
                    If ![Elektrolyseur] = "Reserve" Then
                        ![Elektrolyseur] = rst2!Elektrolyseur
                    End If
                    ![Position] = rst2![Position]
                    ![Einschalt Datum] = rst2![Einschalt Datum]
                    .Update
                End With
            End If
            rst3.Close
            rst2.MoveNext
        Loop
                
    rst2.Close
    
    
    
    
    
     
    'Einlesen der Anodeninstandhaltung-Daten aus der gefundenen Datei
