Private Sub Form_Open(Cancel As Integer)

    Dim rst1 As DAO.Recordset
    Set rst1 = CurrentDb.OpenRecordset("Select * From tblGewDaten ORDER BY IDSprache")
    GewSprache = rst1!IDSprache
    rst1.Close
    
    If GewSprache = 1 Then
        Text8.Caption = "Inspektionsgründe"
        Bezeichnungsfeld110.Caption = "Code"
        Bezeichnungsfeld93.Caption = "Code"
        Bezeichnungsfeld109.Caption = "Inspektionsgrund"
        Bezeichnungsfeld68.Caption = "Inspektionsgrund"
        
        Befehl115.Caption = "Element-verwaltung"
        
    Else
        Text8.Caption = "Reasons for Element Inspection"
        Bezeichnungsfeld110.Caption = "Code"
        Bezeichnungsfeld93.Caption = "Code"
        Bezeichnungsfeld109.Caption = "Reason for Inspection"
        Bezeichnungsfeld68.Caption = "Reason for Inspection"
        
        Befehl115.Caption = "Element Administration"
    End If


End Sub
