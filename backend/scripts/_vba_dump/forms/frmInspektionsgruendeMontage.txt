Public Sub UpdateGruende(CodestringMontageliste As Variant)
    Dim i, j, l, AnzahlCodes, AnzahlGewCodes As Long
    Dim Teilcode(100) As String
    Dim Str1 As String
    Dim rst1 As DAO.Recordset
    
    Set rst1 = Me.RecordsetClone
    
    If rst1.BOF And rst1.EOF Then
        rst1.Close
        Exit Sub
    End If
    
    rst1.MoveFirst
    i = 0
    Do Until rst1.EOF
        i = i + 1
        rst1.MoveNext
    Loop
    rst1.Close
    
    AnzahlCodes = i
    
    i = 1
    Do Until i = AnzahlCodes + 1
        DoCmd.GoToRecord acForm, "frmInspektionsgruendeMontage", acGoTo, i
        Me!Check17 = False
        i = i + 1
    Loop
    
    DoCmd.GoToRecord acForm, "frmInspektionsgruendeMontage", acGoTo, 1
        
    l = Len(CodestringMontageliste)
    If l = 0 Then
        Exit Sub
    End If
            
    i = 0
    Do Until i = 100
        Teilcode(i) = ""
        i = i + 1
    Loop
    AnzahlGewCodes = 1

    i = 1
    j = 1
    Do Until j = l + 1
        Str1 = Mid(CodestringMontageliste, j, 1)
        If Str1 = " " Or Str1 = "," Then
            If Teilcode(i) <> "" Then
                i = i + 1
                AnzahlGewCodes = AnzahlGewCodes + 1
            End If
        Else
            Teilcode(i) = Teilcode(i) + Str1
        End If
        j = j + 1
    Loop
    j = 1
    Do Until j = AnzahlGewCodes + 1
        i = 1
        Do Until i = AnzahlCodes + 1
            DoCmd.GoToRecord acForm, "frmInspektionsgruendeMontage", acGoTo, i
            If Teilcode(j) = Me!Text12 Then
                Me!Check17 = True
            End If
            i = i + 1
        Loop
        j = j + 1
    Loop
    
    DoCmd.GoToRecord acForm, "frmInspektionsgruendeMontage", acGoTo, 1

End Sub
