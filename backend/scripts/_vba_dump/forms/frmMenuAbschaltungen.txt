Private Sub Befehl202_Click()
    Dim rst1 As DAO.Recordset
    Dim dbs As Database
    Dim tdf As TableDef
    
    Set rst1 = CurrentDb.OpenRecordset("Select * From tblGewDaten ORDER BY IDSprache")
    GewSprache = rst1!IDSprache
    rst1.Close
    
    Set dbs = CurrentDb
    Set tdf = dbs.TableDefs!tblAbschaltungsursachen
    If GewSprache = 1 Then
        tdf.Fields!IDSprache.DefaultValue = "1"
    Else
        tdf.Fields!IDSprache.DefaultValue = "2"
    End If
    
    DoCmd.OpenForm "frmAbschaltungsursachen"
    DoCmd.Close acForm, "frmMenuAbschaltungen"
End Sub
