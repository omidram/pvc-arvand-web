Private Sub Befehl191_Click()
Dim rst1, rst2, rst3, rst4, rst5, rst6, rst12 As DAO.Recordset
Dim dbs As Database

'Sprache ermitteln
Set rst12 = CurrentDb.OpenRecordset("Select * From tblGewDaten ORDER BY IDSprache")
GewSprache = rst12!IDSprache
rst12.Close

Set dbs = CurrentDb
dbs.Execute (" Delete * From tblInfoUebertrageneReinstsoleAnalysen ")
Set rst4 = CurrentDb.OpenRecordset("Select * from tblGewDaten Order by DatumReinstsoleanalyseGesamtanlage")
' Set rst1 = CurrentDb.OpenRecordset(" Select * from Normierung Where [I Gesamt] = Null ")
Set rst1 = CurrentDb.OpenRecordset(" SELECT * FROM [Analyse pure brine Ges] WHERE" & _
        " (([Analyse pure brine Ges].Datum)=" & Format$(rst4!DatumReinstsoleanalyseGesamtanlage, "\#m\/d\/yy\#") & ") ")
Set rst5 = CurrentDb.OpenRecordset("Select Bezeichnung from tblElektrolyseurbezeichnung")
Set rst3 = CurrentDb.OpenRecordset("tblInfoUebertrageneReinstsoleAnalysen")
Set rst6 = CurrentDb.OpenRecordset("Select * from [Analyse pure brine]")
If (rst1.BOF And rst1.EOF) Then
    If GewSprache = 1 Then
        MsgBox "Keine Reinstsole-Analysedaten der Gesamtanlage zum gewählten Datum gefunden"
    Else
        MsgBox "No pure brine analysis of total plant for chosen date found"
    End If
    Exit Sub
End If
Do Until rst5.EOF
    Set rst2 = CurrentDb.OpenRecordset(" SELECT * FROM [Analyse pure brine] WHERE" & _
        "((([Analyse pure brine]!Elektrolyseur)='" & rst5![Bezeichnung] & "')" & _
        "AND (([Analyse pure brine]!Datum)=" & Format$(rst4!DatumReinstsoleanalyseGesamtanlage, "\#m\/d\/yy\#") & ")) ")
        
    If (rst2.BOF And rst2.EOF) Then
        rst1.MoveFirst
        rst6.AddNew
        rst6![Elektrolyseur] = rst5![Bezeichnung]
        rst6!Datum = rst1!Datum
        rst6!Zeit = rst1!Zeit
        rst6![NaClO3 Pb] = rst1![NaClO3]
        rst6![Na2SO4 Pb] = rst1![Na2SO4]
        rst6![NaOH Pb] = rst1!NaOH
        rst6![Na2CO3 Pb] = rst1![Na2CO3]
        rst6![NaOCl Pb] = rst1!NaOCl
        rst6!temperature = rst1!temperature
        rst6!pH = rst1!pH
        rst6![density at 20 °C] = rst1![density at 20 °C]
        rst6![NaCl Pb] = rst1![NaCl]
        rst6![Ca +Mg] = rst1![Ca +Mg]
        rst6![Ba] = rst1![Ba]
        rst6![Sr] = rst1![Sr]
        rst6![Ni] = rst1![Ni]
        rst6![Fe] = rst1![Fe]
        rst6![Al] = rst1![Al]
        rst6![SiO2] = rst1![SiO2]
        rst6![i] = rst1![i]
        rst6![Ba] = rst1![Ba]
        rst6![F] = rst1![F]
        rst6![Br] = rst1![Br]
        rst6![Organics] = rst1![Organics]
        rst6.Update
        rst3.AddNew
        rst3!Elektrolyseur = rst5!Bezeichnung
        rst3!Datum = rst1!Datum
        If GewSprache = 1 Then
            rst3!DatenUebertragen = "Analysen übertragen"
        Else
            rst3!DatenUebertragen = "Analytical data transferred"
        End If
        rst3.Update
    Else
        rst3.AddNew
        rst3!Elektrolyseur = rst5!Bezeichnung
        rst3!Datum = rst1!Datum
        If GewSprache = 1 Then
            rst3!DatenUebertragen = "Daten bereits vorhanden"
        Else
            rst3!DatenUebertragen = "Data already existing"
        End If
        rst3.Update
    End If
    subfrmInfoUebertrageneReinstsoleAnalysen.Requery
    rst2.Close
    rst5.MoveNext
Loop
rst1.Close
rst3.Close
rst4.Close
If GewSprache = 1 Then
    MsgBox "Reinstsole-Analysedaten für Einzelelektrolyseure übertragen"
Else
    MsgBox "Pure brine analytical data for single electrolyzers transferred"
End If
End Sub


Private Sub Befehl226_Click()
Dim rst1, rst2, rst3, rst4, rst5, rst6, rst7, rst8, rst12 As DAO.Recordset
Dim dbs As Database

Set dbs = CurrentDb
dbs.Execute (" Delete * From tblInfoUebertrageneReinstsoleAnalysen ")
Set rst4 = CurrentDb.OpenRecordset("Select * from tblGewDaten Order by DatumReinstsoleanalyseTeilanlage")
Set rst1 = CurrentDb.OpenRecordset(" SELECT * FROM [Analyse pure brine Teilanlage] WHERE" & _
        " (([Analyse pure brine Teilanlage].Datum)=" & Format$(rst4!DatumReinstsoleanalyseTeilanlage, "\#m\/d\/yy\#") & ") ")
Set rst5 = CurrentDb.OpenRecordset("Select Bezeichnung from tblElektrolyseurbezeichnung")
Set rst3 = CurrentDb.OpenRecordset("tblInfoUebertrageneReinstsoleAnalysen")
Set rst6 = CurrentDb.OpenRecordset("Select * from [Analyse pure brine]")
Set rst8 = CurrentDb.OpenRecordset(" Select * from tblElektrolyseuranordnung")

'Sprache ermitteln
Set rst12 = CurrentDb.OpenRecordset("Select * From tblGewDaten ORDER BY IDSprache")
GewSprache = rst12!IDSprache
rst12.Close

If (rst1.BOF And rst1.EOF) Then
    If GewSprache = 1 Then
        MsgBox "Keine Reinstsole-Analysedaten der Teilanlage zum gewählten Datum gefunden"
    Else
        MsgBox "No pure brine analysis of train for chosen date found"
    End If
    Exit Sub
End If

If Not rst1.BOF And rst1.EOF Then
    rst1.MoveFirst
End If

Do Until rst1.EOF
    Do Until rst5.EOF
        'Prüfen, ob Daten für einzelne Eletrolyseure bereits vorhanden sind
        Set rst2 = CurrentDb.OpenRecordset(" SELECT * FROM [Analyse pure brine] WHERE" & _
            "((([Analyse pure brine]!Elektrolyseur)='" & rst5![Bezeichnung] & "')" & _
            "AND (([Analyse pure brine]!Datum)=" & Format$(rst4!DatumReinstsoleanalyseTeilanlage, "\#m\/d\/yy\#") & ")) ")
        Set rst7 = CurrentDb.OpenRecordset("SELECT * FROM tblElektrolyseuranordnung WHERE" & _
                   " ( tblElektrolyseuranordnung!Bezeichnung ='" & rst5![Bezeichnung] & "' ) AND " & _
                   " ( tblElektrolyseuranordnung!Teilanlage = '" & rst1!Teilanlage & "') ")
        
        If (rst2.BOF And rst2.EOF) Then
        
            If Not (rst7.BOF And rst7.EOF) Then
                rst6.AddNew
                rst6![Elektrolyseur] = rst5![Bezeichnung]
                rst6!Datum = rst1!Datum
                rst6!Zeit = rst1!Zeit
                rst6![NaClO3 Pb] = rst1![NaClO3 Pb]
                rst6![Na2SO4 Pb] = rst1![Na2SO4 Pb]
                rst6![NaOH Pb] = rst1![NaOH Pb]
                rst6![Na2CO3 Pb] = rst1![Na2CO3 Pb]
                rst6![NaOCl Pb] = rst1![NaOCl Pb]
                rst6!temperature = rst1!temperature
                rst6!pH = rst1!pH
                rst6![density at 20 °C] = rst1![density at 20 °C]
                rst6![NaCl Pb] = rst1![NaCl Pb]
                rst6![Ca +Mg] = rst1![Ca +Mg]
                rst6![Ba] = rst1![Ba]
                rst6![Sr] = rst1![Sr]
                rst6![Ni] = rst1![Ni]
                rst6![Fe] = rst1![Fe]
                rst6![Al] = rst1![Al]
                rst6![SiO2] = rst1![SiO2]
                rst6![i] = rst1![i]
                rst6![Ba] = rst1![Ba]
                rst6![F] = rst1![F]
                rst6![Br] = rst1![Br]
                rst6![Organics] = rst1![Organics]
                rst6.Update
                ' Anzeige aktualisieren
                rst3.AddNew
                rst3!Elektrolyseur = rst5!Bezeichnung
                rst3!Datum = rst1!Datum
                If GewSprache = 1 Then
                    rst3!DatenUebertragen = "Analysen Übertragen"
                Else
                    rst3!DatenUebertragen = "Analytical data transferred"
                End If
                rst3.Update
            End If
        Else
            rst3.AddNew
            rst3!Elektrolyseur = rst5!Bezeichnung
            rst3!Datum = rst1!Datum
            If GewSprache = 1 Then
                rst3!DatenUebertragen = "Daten bereits vorhanden"
            Else
                rst3!DatenUebertragen = "Data already existing"
            End If
            rst3.Update
        End If
        subfrmInfoUebertrageneReinstsoleAnalysen.Requery
        rst2.Close
        rst7.Close
        rst5.MoveNext
    Loop
    rst5.MoveFirst
    rst1.MoveNext
Loop
rst1.Close
rst3.Close
rst4.Close
rst6.Close
rst8.Close
If GewSprache = 1 Then
    MsgBox "Übertragung der Reinstsole-Analysedaten abgeschlossen"
Else
    MsgBox "Transfer of pure brine analytical data completed"
End If

End Sub
