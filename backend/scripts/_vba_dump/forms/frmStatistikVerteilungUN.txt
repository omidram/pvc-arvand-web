Private Sub VerteilungBerechnen()

Dim rst1, rst2, rst3, rst12 As DAO.Recordset
Dim Umin, Umax, Schrittweite, UG, OG As Double
Dim dbs As Database

Set dbs = CurrentDb
dbs.Execute (" Delete * From tblVerteilungUnElemente ")
dbs.Execute (" Delete * From tblVerteilungUnElementeDetails ")
dbs.Execute (" Delete * From tblKlassenVerteilungUnElemente ")

'Grenzen und Schrittweite einlesen
    Set rst12 = CurrentDb.OpenRecordset("Select * From tblGewDaten ORDER BY IDSprache")
    Umin = rst12!UminVerteilungUn
    Umax = rst12!UmaxVerteilungUn
    Schrittweite = rst12!SchrittweiteVerteilungUn
    rst12.Close
    
'Klassen berechnen
    Set rst1 = CurrentDb.OpenRecordset("Select * From tblKlassenVerteilungUnElemente")
    rst1.AddNew
    rst1.Untergrenze = 0
    Umin = Round(Umin, 3)
    rst1.Obergrenze = Round(Umin, 3)
    rst1.Klasse = "<=" + Str(Umin)
    rst1.Update
    UG = Umin
    OG = Round(Umin + Schrittweite, 3)
    
    Do Until OG > Umax
        rst1.AddNew
        rst1.Untergrenze = Round(UG, 3)
        rst1.Obergrenze = Round(OG, 3)
        rst1.Klasse = Str(UG) + " -" + Str(OG)
        rst1.Update
        UG = Round(OG, 3)
        OG = Round(OG + Schrittweite, 3)
    Loop
    
    rst1.AddNew
    rst1.Untergrenze = Umax
    rst1.Obergrenze = 100
    rst1.Klasse = ">" + Str(Umax)
    rst1.Update
    rst1.Close
    
'Elemente Klassen zuordnen
    Set rst1 = CurrentDb.OpenRecordset("Select * From qryStatistikElementeUnVerteilung")
    Set rst2 = CurrentDb.OpenRecordset("Select * From tblKlassenVerteilungUnElemente")
    Set rst3 = CurrentDb.OpenRecordset("Select * From tblVerteilungUnElementeDetails")
    
    Do Until rst1.EOF
        rst2.MoveFirst
        rst3.AddNew
        rst3!Elektrolyseur = rst1!Elektrolyseur
        rst3!Position = rst1!Position
        rst3!Datum = rst1!Datum
        rst3!Zeit = rst1!Zeit
        rst3![I Gesamt] = rst1![I Gesamt]
        rst3!i = rst1!i
        rst3![t An] = rst1![t An]
        rst3![t Ka] = rst1![t Ka]
        rst3!tm = rst1!tm
        rst3!Cc = rst1!Cc
        rst3!iRef = rst1!iRef
        rst3!Ui = rst1!Ui
        rst3!Un = rst1!Un
        Do Until rst2.EOF
            If (rst1!Un > rst2.Untergrenze) And (rst1!Un <= rst2.Obergrenze) Then
                rst3!Klasse = rst2!Klasse
            End If
            rst2.MoveNext
        Loop
        rst3.Update
        rst1.MoveNext
    Loop
    
    rst3.Close
    rst2.Close
    rst1.Close


End Sub
