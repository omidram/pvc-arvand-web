Private Sub Form_Open(Cancel As Integer)
    Dim rst1 As DAO.Recordset
    Set rst1 = CurrentDb.OpenRecordset("Select * From tblGewDaten ORDER BY IDSprache")
    GewSprache = rst1!IDSprache
    rst1.Close
    If GewSprache = 1 Then
        Bezeichnungsfeld168.Caption = "Elemente ohne Laufzeit"
        Schaltfläche27.Caption = "Hauptmenu"
        Befehl154.Caption = "Such-funktionen"
        
    Else
        Bezeichnungsfeld168.Caption = "Elements without DOL"
        Schaltfläche27.Caption = "Main Menu"
        Befehl154.Caption = "Search"
    End If
End Sub
