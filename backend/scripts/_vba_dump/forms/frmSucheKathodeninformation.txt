Private Sub Form_Open(Cancel As Integer)
    Dim rst1 As DAO.Recordset
    Set rst1 = CurrentDb.OpenRecordset("Select * From tblGewDaten ORDER BY IDSprache")
    GewSprache = rst1!IDSprache
    rst1.Close
    If GewSprache = 1 Then
        Befehl59.Caption = "Anzeige aktualisieren"
        Befehl80.Caption = "Hauptmenu"
        Schaltfläche43.Caption = "Suchfunktionen"
        Bezeichnungsfeld83.Caption = "Kathodendetails"
        Bezeichnungsfeld81.Caption = "Kathodengeschichte"
        Bezeichnungsfeld85.Caption = "Kathodeninstandhaltung"
        Bezeichnungsfeld87.Caption = "Kathodenrecoating"
        Bezeichnungsfeld89.Caption = "Kathodencoatingprüfung"
    
    Else
        Befehl59.Caption = "Update Display"
        Befehl80.Caption = "Main Menu"
        Schaltfläche43.Caption = "Search"
        Bezeichnungsfeld83.Caption = "Kathode Details"
        Bezeichnungsfeld81.Caption = "Cathode History"
        Bezeichnungsfeld85.Caption = "Cathode Maintenance"
        Bezeichnungsfeld87.Caption = "Cathode Recoating"
        Bezeichnungsfeld89.Caption = "Cathode Coating Inspection"

    End If
    
    Me!Befehl59.SetFocus

End Sub
