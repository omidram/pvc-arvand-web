Private Sub Befehl224_Click()
    Dim rst1 As DAO.Recordset
    Dim dbs As Database
    Dim tdf As TableDef
    
    Set rst1 = CurrentDb.OpenRecordset("Select * From tblGewDaten ORDER BY IDSprache")
    GewSprache = rst1!IDSprache
    rst1.Close
    
    Set dbs = CurrentDb
    Set tdf = dbs.TableDefs!tblInspektionsgruende
    If GewSprache = 1 Then
        tdf.Fields!IDSprache.DefaultValue = "1"
    Else
        tdf.Fields!IDSprache.DefaultValue = "2"
    End If
    
    DoCmd.OpenForm "frmInspektionsgruende"
    DoCmd.Close acForm, "frmZellenverwaltung"
End Sub
