Private Sub Form_Open(Cancel As Integer)

    Dim rst1 As DAO.Recordset
    Set rst1 = CurrentDb.OpenRecordset("Select * From tblGewDaten ORDER BY IDSprache")
    GewSprache = rst1!IDSprache
    rst1.Close
    If GewSprache = 1 Then
    
        Me![Bezeichnungsfeld66].Caption = "Versanddatum"
        Me![Bezeichnungsfeld68].Caption = "Hersteller"
        Me![Bezeichnungsfeld93].Caption = "Anodennummer"
        Me![Bezeichnungsfeld111].Caption = "Rückkehrdatum"
        Bezeichnungsfeld124.Caption = "Bemerkung"
        
    Else
    
        Me![Bezeichnungsfeld66].Caption = "Shipping Date"
        Me![Bezeichnungsfeld68].Caption = "Manufacturer"
        Me![Bezeichnungsfeld93].Caption = "Anode No."
        Me![Bezeichnungsfeld111].Caption = "Return Date"
        Bezeichnungsfeld124.Caption = "Remark"
                
    End If


End Sub
