Private Sub Form_Open(Cancel As Integer)
    Dim rst1 As DAO.Recordset
    Set rst1 = CurrentDb.OpenRecordset("Select * From tblGewDaten ORDER BY IDSprache")
    GewSprache = rst1!IDSprache
    rst1.Close
    If GewSprache = 1 Then
        Bezeichnungsfeld55.Caption = "Elektrolyseur"
        Bezeichnungsfeld56.Caption = "Block"
        Bezeichnungsfeld57.Caption = "Anfangsposition"
        Bezeichnungsfeld58.Caption = "Endposition"
        Bezeichnungsfeld60.Caption = "Teilanlage"
        Bezeichnungsfeld61.Caption = "Transformator"
        Bezeichnungsfeld62.Caption = "Gleichrichter"
        
    Else
        Bezeichnungsfeld55.Caption = "Electrolyzer"
        Bezeichnungsfeld56.Caption = "Rack"
        Bezeichnungsfeld57.Caption = "Start Position"
        Bezeichnungsfeld58.Caption = "End Position"
        Bezeichnungsfeld60.Caption = "Train"
        Bezeichnungsfeld61.Caption = "Transformer"
        Bezeichnungsfeld62.Caption = "Rectifier"
        
    End If

End Sub
