Private Sub Form_Open(Cancel As Integer)

    Dim rst1 As DAO.Recordset
    Set rst1 = CurrentDb.OpenRecordset("Select * From tblGewDaten ORDER BY IDSprache")
    GewSprache = rst1!IDSprache
    rst1.Close
    If GewSprache = 1 Then
        
        Me![Cathode Number].Caption = "Kathodennummer"
        Me![Shipping Date].Caption = "Versanddatum"
        Me![Return Date].Caption = "Rückkehrdatum"
        Me![Manufacturer].Caption = "Hersteller"
        Bezeichnungsfeld10.Caption = "Bemerkung"
        
    Else
    
        Me![Cathode Number].Caption = "Cathode No."
        Me![Shipping Date].Caption = "Shipping Date"
        Me![Return Date].Caption = "Return Date"
        Me![Manufacturer].Caption = "Manufacturer"
        Bezeichnungsfeld10.Caption = "Remark"
    
    End If

End Sub
