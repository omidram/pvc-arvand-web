Private Sub Form_Open(Cancel As Integer)

    Dim rst1 As DAO.Recordset
    Set rst1 = CurrentDb.OpenRecordset("Select * From tblGewDaten ORDER BY IDSprache")
    GewSprache = rst1!IDSprache
    rst1.Close
    If GewSprache = 1 Then
        Bezeichnungsfeld110.Caption = "Reparaturdatum"
        Bezeichnungsfeld105.Caption = "Membran-Nr."
        Bezeichnungsfeld101.Caption = "Art der Reparatur"
        
    Else
        Bezeichnungsfeld110.Caption = "Date of Repair"
        Bezeichnungsfeld105.Caption = "Membrane No."
        Bezeichnungsfeld101.Caption = "Repair Work"
        
    End If

End Sub
