Private Sub UpdateKategorie()
    Dim GewKategorie, GewCode As Variant
    Dim rst2 As DAO.Recordset
    
    Set rst2 = CurrentDb.OpenRecordset(" SELECT * FROM qryAbschaltungsursachen WHERE" & _
    "(((qryAbschaltungsursachen.Abschaltungsursache)='" & Me!Kombinationsfeld118 & "'))  ")
        
    If rst2.BOF And rst2.EOF Then
        rst2.Close
        Exit Sub
    End If
    rst2.MoveFirst
    GewKategorie = rst2!Kategorie
    rst2.Close
    Me!Text126 = GewKategorie
    
End Sub
